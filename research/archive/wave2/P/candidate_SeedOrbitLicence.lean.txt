/-! ### §A — the generic step: generator-closed families carry the whole word orbit -/

/-- A family closed under precomposition with every generator and its inverse. -/
def GenClosed (S : Set (V ≃ᵃ[ℝ] V)) (A : Set (V →ᵃ[ℝ] ℝ)) : Prop :=
  ∀ e ∈ A, ∀ s ∈ S, e.comp s.toAffineMap ∈ A ∧ e.comp s.symm.toAffineMap ∈ A

/-- P-1. A generator-closed family containing the seed contains its whole word orbit.
Sketch: `Subgroup.closure_induction` on `P g := ∀ e ∈ A, e ∘ g⁻¹ ∈ A ∧ e ∘ g ∈ A`;
mul uses `(g*k) x = g (k x)` (`mul_apply'`, ON:69): `e ∘ (gk)⁻¹ = (e ∘ k⁻¹) ∘ g⁻¹`, `e ∘ gk = (e ∘ g) ∘ k`. -/
theorem seedOrbitAvailable_of_genClosed {S : Set (V ≃ᵃ[ℝ] V)} {r : V →ᵃ[ℝ] ℝ}
    {A : Set (V →ᵃ[ℝ] ℝ)} (hr : r ∈ A) (hA : GenClosed S A) :
    SeedOrbitAvailable (words S) r A := sorry

/-- P-2, the circularity certificate. V4′ on the words is EXACTLY the existence of a
generator-closed subfamily containing the seed (witness: `seedOrbit (words S) r`). -/
theorem seedOrbitAvailable_words_iff {S : Set (V ≃ᵃ[ℝ] V)} {r : V →ᵃ[ℝ] ℝ}
    {avail : Set (V →ᵃ[ℝ] ℝ)} :
    SeedOrbitAvailable (words S) r avail ↔ ∃ A ⊆ avail, r ∈ A ∧ GenClosed S A := sorry

/-! ### §B — the minimal operational extension: one system, no composite -/

/-- An operational theory over a body: available effects and available transformations.
The two consistency fields are not used by P-4; they make `SeqClosed` coherent (control E5). -/
structure BodyOperationalTheory (Ω : Set V) where
  eff : Set (V →ᵃ[ℝ] ℝ)
  trans : Set (V ≃ᵃ[ℝ] V)
  eff_effect : ∀ e ∈ eff, IsEffectOn Ω e
  trans_maps : ∀ T ∈ trans, ∀ x ∈ Ω, T x ∈ Ω

/-- SEQ (named hypothesis, NOT a field): a test performed after an available transformation is
an available test. Schrödinger reading: the state is transformed first, then read. -/
def SeqClosed {Ω : Set V} (Θ : BodyOperationalTheory Ω) : Prop :=
  ∀ e ∈ Θ.eff, ∀ T ∈ Θ.trans, e.comp T.toAffineMap ∈ Θ.eff

/-- TRANS± (named hypothesis): the drive's flow, `J` and `J⁻¹` are available transformations.
Not contained in `ElementaryDrivability` (KF:264), which carries no availability. -/
def DriveAvailable {Ω : Set V} (Θ : BodyOperationalTheory Ω) (D : ElementaryDrivability Ω) :
    Prop :=
  (∀ t, D.flow t ∈ Θ.trans) ∧ D.J ∈ Θ.trans ∧ D.J.symm ∈ Θ.trans

/-- P-3. Consistency: SEQ never leaves the effects (uses `trans_maps`, `eff_effect`). -/
theorem isEffectOn_comp_of_trans {Ω : Set V} (Θ : BodyOperationalTheory Ω) {e : V →ᵃ[ℝ] ℝ}
    {T : V ≃ᵃ[ℝ] V} (he : e ∈ Θ.eff) (hT : T ∈ Θ.trans) :
    IsEffectOn Ω (e.comp T.toAffineMap) := sorry
